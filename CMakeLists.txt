cmake_minimum_required(VERSION 3.21)
project(SonyCameraControl VERSION 1.3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SONY_SDK_ROOT "" CACHE PATH "Extracted SimpleCli SDK directory (app/CRSDK and external/crsdk)")
if(NOT EXISTS "${SONY_SDK_ROOT}/app/CRSDK/CameraRemote_SDK.h")
  message(FATAL_ERROR "Set SONY_SDK_ROOT to the extracted platform-specific SimpleCli directory")
endif()

# Explicitly opt in to build and run this tool: it operates a real camera.
add_executable(hardware_validation EXCLUDE_FROM_ALL Tests/HardwareValidation.cpp)
target_include_directories(hardware_validation PRIVATE ThirdParty)
target_include_directories(hardware_validation SYSTEM PRIVATE "${SONY_SDK_ROOT}/app/CRSDK")
target_link_libraries(hardware_validation PRIVATE ${CMAKE_DL_LIBS})
add_dependencies(hardware_validation camera_bridge)
set_target_properties(hardware_validation PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/native")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(hardware_validation PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/native")
endforeach()
if(MSVC)
  target_compile_definitions(hardware_validation PRIVATE UNICODE _UNICODE)
  target_compile_options(hardware_validation PRIVATE /utf-8)
endif()
add_custom_command(TARGET hardware_validation POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${SONY_SDK_ROOT}/external/crsdk" "$<TARGET_FILE_DIR:hardware_validation>")
find_package(Threads REQUIRED)
find_library(SONY_CORE NAMES Cr_Core PATHS "${SONY_SDK_ROOT}/external/crsdk" NO_DEFAULT_PATH REQUIRED)
add_library(camera_bridge SHARED Sources/Portable/CameraBridge.cpp)
target_include_directories(camera_bridge PRIVATE Sources ThirdParty)
target_include_directories(camera_bridge SYSTEM PRIVATE "${SONY_SDK_ROOT}/app/CRSDK")
target_link_libraries(camera_bridge PRIVATE "${SONY_CORE}" Threads::Threads)
set_target_properties(camera_bridge PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/native"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/native"
  BUILD_RPATH "$ORIGIN;$ORIGIN/CrAdapter;${SONY_SDK_ROOT}/external/crsdk"
  INSTALL_RPATH "$ORIGIN;$ORIGIN/CrAdapter")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(camera_bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/native")
endforeach()
if(MSVC)
  target_compile_definitions(camera_bridge PRIVATE UNICODE _UNICODE NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_compile_options(camera_bridge PRIVATE /utf-8 /W4 /EHsc)
  set_property(TARGET camera_bridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(camera_bridge PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS camera_bridge RUNTIME DESTINATION native LIBRARY DESTINATION native)
install(DIRECTORY "${SONY_SDK_ROOT}/external/crsdk/" DESTINATION native
  PATTERN "*.lib" EXCLUDE PATTERN "*.a" EXCLUDE)

include(CTest)
if(BUILD_TESTING)
  add_executable(portable_bridge_tests Tests/PortableCameraBridgeTests.cpp)
  target_include_directories(portable_bridge_tests PRIVATE Sources ThirdParty)
  target_include_directories(portable_bridge_tests SYSTEM PRIVATE "${SONY_SDK_ROOT}/app/CRSDK")
  target_link_libraries(portable_bridge_tests PRIVATE "${SONY_CORE}" Threads::Threads)
  if(MSVC)
    target_compile_definitions(portable_bridge_tests PRIVATE UNICODE _UNICODE NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(portable_bridge_tests PRIVATE /utf-8 /UNDEBUG)
    set_property(TARGET portable_bridge_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_custom_command(TARGET portable_bridge_tests POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SONY_SDK_ROOT}/external/crsdk/Cr_Core.dll" "$<TARGET_FILE_DIR:portable_bridge_tests>/Cr_Core.dll")
  else()
    target_compile_options(portable_bridge_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME portable_bridge COMMAND portable_bridge_tests)
  set_tests_properties(portable_bridge PROPERTIES TIMEOUT 120)
endif()
